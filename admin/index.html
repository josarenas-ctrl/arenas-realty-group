return new Response(html, { headers: { "Content-Type": "text/html" }, }); } // --------------------------------------------------------------- // NUEVO: página + acción para autorizar asesores como colaboradores // --------------------------------------------------------------- if (url.pathname === "/invitar") { if (request.method === "GET") { return new Response(formularioInvitar(), { headers: { "Content-Type": "text/html;
